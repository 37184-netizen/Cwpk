        // --- LOGIC: Group words by sorted characters (Anagrams) ---
        const getAlphagram = (word) => word.split('').sort().join('');

        const createQuestions = (wordList) => {
          const groups = {};
          
          wordList.forEach(word => {
            const cleanWord = word.trim().toUpperCase();
            if (cleanWord.length === 0) return;

            const alpha = getAlphagram(cleanWord);
            if (!groups[alpha]) {
              groups[alpha] = [];
            }
            if (!groups[alpha].includes(cleanWord)) {
              groups[alpha].push(cleanWord);
            }
          });

          return Object.keys(groups).map(alpha => ({
            scramble: alpha,
            answers: groups[alpha]
          })).filter(q => q.answers.length > 0);
        };

            [arr[i], arr[j]] = [arr[j], arr[i]];

          return arr;

                          <div>
                            <div className="flex items-center gap-2 mb-2">
                               <span className="bg-white/20 text-white text-[10px] px-2 py-0.5 rounded-full uppercase font-bold border border-white/20">Recommended</span>
                            </div>
                            <h3 className="text-4xl font-black italic tracking-wide drop-shadow-md">BINGO 7</h3>
                            <p className="text-pink-100 text-sm mt-1 font-medium">Standard Vocabulary</p>
                          </div>
                          <div className="bg-white/20 p-3 rounded-2xl backdrop-blur-sm">
                            <Trophy size={48} className="text-yellow-300 drop-shadow-sm" />
                          </div>
                        </div>
                      </button>

                      {/* Grid 2-6 Buttons */}
                      <div className="grid grid-cols-3 gap-3">
                        {[2, 3, 4, 5, 6].map(num => (
                          <button 
                            key={num}
                            onClick={() => startGame(num)}
                            className="bg-white border-2 border-slate-100 rounded-2xl p-3 flex flex-col items-center justify-center gap-1 transition-all h-28 shadow-sm hover:border-blue-400 hover:shadow-md hover:-translate-y-1 group"
                          >
                            <span className="text-3xl font-black text-slate-700 group-hover:text-blue-600 transition-colors">{num}</span>
                            <span className="text-[10px] font-bold text-slate-400 uppercase group-hover:text-blue-400">Letters</span>
                          </button>
                        ))}
                        
                        {/* Mix Button (Dark Blue) */}
                         <button 
                            onClick={() => startGame('mix')}
                            className="bg-slate-800 text-white border-b-4 border-slate-950 active:border-t-4 active:border-b-0 rounded-2xl p-3 flex flex-col items-center justify-center gap-1 transition-all h-28 shadow-lg hover:bg-slate-900 group"
                          >
                            <Shuffle size={28} className="text-pink-400 group-hover:rotate-180 transition-transform duration-500" />
                            <span className="text-[10px] font-bold text-slate-400 uppercase mt-1">Mix All</span>
                          </button>
                      </div>

                      <div className="mt-2 bg-blue-50/80 p-4 rounded-2xl border border-blue-100 flex items-center gap-3 glass-panel">
                        <Info className="text-blue-500 shrink-0" size={20} />
                        <p className="text-xs text-blue-800 font-medium">
                          อัปเดตคำศัพท์ <b>ZEAL, EW</b> และระบบตัดคำใหม่ (Bold) ออกแล้ว เพื่อความแม่นยำในการแข่งขัน
                        </p>
                      </div>
                    </div>
                  )}

                  {/* --- PLAYING STATE --- */}
                  {gameState === 'playing' && (
                    <div className="flex flex-col h-full animate-in fade-in zoom-in duration-300">
                      {/* Score Board */}
                      <div className="bg-white/60 backdrop-blur-md px-6 py-4 border-b border-slate-100 flex justify-between items-center sticky top-0 z-10">
                          <div className="flex flex-col">
                            <span className="text-[10px] font-bold text-slate-400 uppercase tracking-widest mb-1">Score</span>
                            <span className="text-4xl font-black text-transparent bg-clip-text bg-gradient-to-r from-blue-600 to-pink-500 leading-none drop-shadow-sm">{score}</span>
                          </div>
                          
                          <div className="flex flex-col items-end">
                            <span className="text-[10px] font-bold text-slate-400 uppercase tracking-widest mb-1">Puzzle</span>
                            <span className="text-2xl font-bold text-slate-700 leading-none">{currentQIndex + 1}<span className="text-sm text-slate-300 font-medium">/{questions.length}</span></span>
                          </div>
                      </div>

                      {/* Game Area */}
                      <div className="flex-1 flex flex-col items-center p-4 w-full max-w-md mx-auto">
                        
                        {/* Tiles */}
                        <div className="mb-6 w-full">
                          <div className="flex flex-wrap justify-center gap-2">
                            {currentScrambleTiles.map((char, idx) => (
                              <div 
                                key={`${char}-${idx}`}
                                className="relative w-11 h-11 sm:w-14 sm:h-14 bg-gradient-to-b from-[#f8ebc7] to-[#e4c988] rounded-xl shadow-[0_4px_0_#c4a355] flex items-center justify-center text-xl sm:text-2xl font-bold text-[#4a3b22] select-none animate-in zoom-in border border-[#e6d5a7]"
                                style={{animationDelay: `${idx * 50}ms`}}
                              >
                                {char}
                                <span className="absolute bottom-0.5 right-1 text-[8px] opacity-70 font-semibold">
                                  {getTileScore(char)}
                                </span>
                              </div>
                            ))}
                          </div>
                          
                          <div className="flex justify-center mt-5">
                             <button 
                              onClick={() => setCurrentScrambleTiles(scrambleWord(questions[currentQIndex].scramble))}
                              className="text-xs font-bold text-slate-500 hover:text-blue-600 flex items-center gap-1.5 bg-white border border-slate-200 px-4 py-2 rounded-full transition-all hover:shadow-md hover:border-blue-200"
                            >
                              <RefreshCw size={14} /> สลับตัวอักษร
                            </button>
                          </div>
                        </div>

                        {/* Answer Slots */}
                        <div className="w-full mb-6 space-y-2">
                           <div className="flex justify-between items-end mb-2 px-2">
                              <span className="text-xs font-bold text-slate-400 uppercase tracking-wider flex items-center gap-1">
                                <List size={12} /> Found ({foundWords.length}/{questions[currentQIndex].answers.length})
                              </span>
                           </div>
                           
                           <div className="grid grid-cols-1 gap-2.5 max-h-[220px] overflow-y-auto pr-1 custom-scrollbar">
                             {questions[currentQIndex].answers.map((ans, idx) => {
                               const isFound = foundWords.includes(ans);
                               return (
                                 <div 
                                   key={idx} 
                                   className={`h-12 rounded-xl flex items-center px-4 font-mono text-lg tracking-widest transition-all duration-500 border-2 ${
                                     isFound 
                                       ? 'bg-green-50 border-green-200 text-green-700 shadow-sm' 
                                       : 'bg-slate-50 border-dashed border-slate-200 text-slate-300 justify-center'
                                   }`}
                                 >
                                   {isFound ? (
                                     <div className="flex justify-between w-full items-center animate-in slide-in-from-left-2 fade-in">
                                       <span className="font-bold">{ans}</span>
                                       <Check size={18} className="text-green-500" />
                                     </div>
                                   ) : (
                                     <span>{'•'.repeat(ans.length)}</span>
                                   )}
                                 </div>
                               );
                             })}
                           </div>
                        </div>

                        {/* Feedback Toast */}
                        <div className={`h-10 mb-2 flex items-center justify-center transition-all duration-300 ${message.text ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-2'}`}>
                            <span className={`px-5 py-2 rounded-full text-sm font-bold flex items-center gap-2 shadow-lg ${
                              message.type === 'success' ? 'text-white bg-gradient-to-r from-green-500 to-emerald-600' : 
                              message.type === 'info' ? 'text-white bg-gradient-to-r from-blue-500 to-indigo-600' :
                              'text-white bg-gradient-to-r from-red-500 to-rose-600'
                            }`}>
                              {message.type === 'success' && <Star size={14} fill="white" />}
                              {message.text}
                            </span>
                        </div>

                        {/* Input Area */}
                        <form onSubmit={checkAnswer} className="w-full relative group">
                          <input
                            ref={inputRef}
                            type="text"
                            value={userGuess}
                            onChange={(e) => setUserGuess(e.target.value)}
                            className="w-full bg-white text-center text-2xl font-bold tracking-[0.15em] uppercase py-4 px-12 border-2 border-slate-200 rounded-2xl focus:border-pink-500 focus:ring-4 focus:ring-pink-100 outline-none transition-all shadow-sm disabled:opacity-50 disabled:bg-slate-100 text-slate-800"
                            placeholder="พิมพ์คำตอบ..."
                            disabled={foundWords.length === questions[currentQIndex].answers.length}
                          />
                          <div className="absolute left-5 top-0 bottom-0 flex items-center pointer-events-none">
                            <Hash size={20} className="text-slate-300 group-focus-within:text-pink-400 transition-colors" />
                          </div>
                          <button 
                            type="submit"
                            disabled={foundWords.length === questions[currentQIndex].answers.length}
                            className="absolute right-2 top-2 bottom-2 bg-gradient-to-r from-blue-600 to-indigo-600 hover:from-blue-500 hover:to-indigo-500 disabled:from-slate-300 disabled:to-slate-400 text-white w-12 rounded-xl flex items-center justify-center shadow-md transition-all active:scale-95"
                          >
                            <ChevronRight size={24} />
                          </button>
                        </form>

                        {/* Footer Actions */}
                        <div className="w-full mt-4 flex justify-end">
                          <button 
                            onClick={handleSkip}
                            className="text-xs font-bold text-slate-400 hover:text-red-500 flex items-center gap-1.5 py-2 px-3 hover:bg-red-50 rounded-lg transition-colors"
                          >
                            ยอมแพ้ / ดูเฉลย <Play size={10} fill="currentColor" />
                          </button>
                        </div>

                      </div>
                    </div>
                  )}

                  {/* --- SUMMARY STATE --- */}
                  {gameState === 'summary' && (
                    <div className="h-full flex flex-col items-center justify-center p-8 text-center animate-in zoom-in duration-300">
                      <div className="relative mb-8">
                         <div className="absolute inset-0 bg-yellow-400 blur-3xl opacity-20 rounded-full animate-pulse"></div>
                         <Trophy size={80} className="text-yellow-400 drop-shadow-lg relative z-10" />
                      </div>
                      
                      <h2 className="text-4xl font-black text-slate-800 mb-2">จบเกม!</h2>
                      <p className="text-slate-500 mb-8 font-medium">นี่คือผลคะแนนของคุณ</p>

                      <div className="w-full max-w-xs bg-white rounded-3xl shadow-xl border border-slate-100 overflow-hidden mb-8">
                         <div className="p-8 border-b border-slate-50 bg-gradient-to-br from-blue-600 to-indigo-700 text-white relative overflow-hidden">
                            <div className="absolute top-0 right-0 w-32 h-32 bg-white/10 rounded-full blur-2xl translate-x-10 -translate-y-10"></div>
                            <span className="text-xs font-bold uppercase tracking-widest opacity-80 relative z-10">Total Score</span>
                            <div className="text-6xl font-black mt-2 relative z-10 tracking-tight">{score}</div>
                         </div>
                         <div className="p-5 bg-white flex justify-between text-center">
                            <div className="w-1/2 border-r border-slate-100">
                                <div className="text-[10px] text-slate-400 font-bold uppercase">Puzzles</div>
                                <div className="text-xl font-bold text-slate-800">{currentQIndex}</div>
                            </div>
                             <div className="w-1/2">
                                <div className="text-[10px] text-slate-400 font-bold uppercase">Mode</div>
                                <div className="text-xl font-bold text-slate-800">{selectedMode === 'mix' ? 'Mix' : `${selectedMode} Letters`}</div>
                            </div>
                         </div>
                      </div>

                      <button 
                        onClick={() => setGameState('menu')}
                        className="w-full max-w-xs py-4 bg-slate-800 text-white rounded-2xl font-bold hover:bg-black transition-colors flex items-center justify-center gap-2 shadow-lg"
                      >
                        <RefreshCw size={20} />
                        เล่นใหม่อีกครั้ง
                      </button>
                    </div>
                  )}
                </div>
              </div>
            </div>
          );
        }

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<App />);
    </script>
